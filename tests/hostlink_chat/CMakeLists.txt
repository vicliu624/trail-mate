cmake_minimum_required(VERSION 3.16)
project(hostlink_chat LANGUAGES CXX)
enable_testing()
option(HOSTLINK_CHAT_SANITIZERS "Enable address/undefined behaviour sanitizers" OFF)
if(HOSTLINK_CHAT_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_executable(chat_submission test_submission.cpp)
target_compile_features(chat_submission PRIVATE cxx_std_17)
target_compile_options(chat_submission PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_include_directories(chat_submission PRIVATE ../../modules/core_hostlink/include)
add_test(NAME chat_submission COMMAND chat_submission)
add_executable(chat_journal test_journal.cpp)
target_compile_features(chat_journal PRIVATE cxx_std_17)
target_compile_options(chat_journal PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_include_directories(chat_journal PRIVATE ../../modules/core_hostlink/include ../../modules/core_geocaching/include)
add_test(NAME chat_journal COMMAND chat_journal)
add_executable(chat_runtime test_runtime.cpp ../../modules/ui_shared/src/ui/presentation_sources/team_chat_action_sink.cpp)
target_compile_features(chat_runtime PRIVATE cxx_std_17)
target_compile_options(chat_runtime PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_compile_definitions(chat_runtime PRIVATE TRAIL_MATE_USB_HOSTLINK_AUTOSTART=1)
target_include_directories(chat_runtime PRIVATE fakes ../../modules/core_hostlink/include ../../modules/core_chat/include ../../modules/core_team/include ../../modules/core_sys/include ../../modules/ui_shared/include ../../modules/ui_presentation/include ../../modules/core_geocaching/include ../../platform/esp/arduino_common/include)
add_test(NAME chat_runtime COMMAND chat_runtime)
