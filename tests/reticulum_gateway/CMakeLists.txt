cmake_minimum_required(VERSION 3.20)
project(reticulum_gateway LANGUAGES CXX)
enable_testing()
set(REPO "${CMAKE_CURRENT_LIST_DIR}/../..")
set(SOURCE_FILE "${REPO}/platform/esp/arduino_common/src/chat/infra/reticulum/reticulum_interfaces.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SOURCE_FILE}")
file(READ "${SOURCE_FILE}" SOURCE)
function(extract_between START END OUTPUT)
    string(FIND "${SOURCE}" "${START}" FIRST)
    if(FIRST LESS 0)
        message(FATAL_ERROR "Gateway extraction start changed")
    endif()
    string(SUBSTRING "${SOURCE}" ${FIRST} -1 TAIL)
    string(FIND "${TAIL}" "${END}" LENGTH)
    if(LENGTH LESS 0)
        message(FATAL_ERROR "Gateway extraction end changed")
    endif()
    string(SUBSTRING "${TAIL}" 0 ${LENGTH} BODY)
    set(${OUTPUT} "${BODY}" PARENT_SCOPE)
endfunction()
extract_between("void WifiGatewayReticulumInterface::stop()" "bool WifiGatewayReticulumInterface::connected()" STOP)
extract_between("bool WifiGatewayReticulumInterface::ensureSocket()" "void WifiGatewayReticulumInterface::readAvailable()" CONNECT)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/gateway_actual.inc" "${STOP}\n${CONNECT}")
add_executable(reticulum_gateway test_gateway.cpp)
target_compile_features(reticulum_gateway PRIVATE cxx_std_17)
target_include_directories(reticulum_gateway PRIVATE "${REPO}/modules/core_chat/include" "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
    target_compile_options(reticulum_gateway PRIVATE /UNDEBUG)
else()
    target_compile_options(reticulum_gateway PRIVATE -UNDEBUG -Wall -Wextra)
endif()
add_test(NAME reticulum_gateway COMMAND reticulum_gateway)
