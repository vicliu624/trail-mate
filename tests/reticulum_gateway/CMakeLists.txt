set(REPO "${CMAKE_CURRENT_LIST_DIR}/../..")
set(SOURCE_FILE "${REPO}/platform/esp/arduino_common/src/chat/infra/reticulum/reticulum_interfaces.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SOURCE_FILE}")
file(READ "${SOURCE_FILE}" SOURCE)
function(extract_between START END OUTPUT)
    string(FIND "${SOURCE}" "${START}" FIRST)
    if(FIRST LESS 0)
        message(FATAL_ERROR "Gateway extraction start changed")
    endif()
    string(SUBSTRING "${SOURCE}" ${FIRST} -1 TAIL)
    string(FIND "${TAIL}" "${END}" LENGTH)
    if(LENGTH LESS 0)
        message(FATAL_ERROR "Gateway extraction end changed")
    endif()
    string(SUBSTRING "${TAIL}" 0 ${LENGTH} BODY)
    set(${OUTPUT} "${BODY}" PARENT_SCOPE)
endfunction()
extract_between("void WifiGatewayReticulumInterface::stop()" "bool WifiGatewayReticulumInterface::connected()" STOP)
extract_between("bool WifiGatewayReticulumInterface::ensureSocket()" "void WifiGatewayReticulumInterface::readAvailable()" CONNECT)
extract_between("void WifiGatewayReticulumInterface::maintain()" "bool WifiGatewayReticulumInterface::isConfigured()" MAINTAIN)
extract_between("void ReticulumInterfaceSet::maintain()" "bool ReticulumInterfaceSet::hasReadyInterface()" SELECT)
extract_between("void WifiGatewayReticulumInterface::readAvailable()" "void WifiGatewayReticulumInterface::feedHdlcByte(" RECEIVE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/gateway_actual.inc" "${STOP}\n${CONNECT}\n${MAINTAIN}\n${SELECT}\n${RECEIVE}")
add_executable(reticulum_gateway test_gateway.cpp)
target_compile_features(reticulum_gateway PRIVATE cxx_std_17)
target_include_directories(reticulum_gateway PRIVATE "${REPO}/modules/core_chat/include" "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
    target_compile_options(reticulum_gateway PRIVATE /UNDEBUG)
else()
    target_compile_options(reticulum_gateway PRIVATE -UNDEBUG -Wall -Wextra)
endif()
add_test(NAME reticulum_gateway COMMAND reticulum_gateway)

# The firmware already depends on rweather/Crypto. Host verification uses the
# same SHA-256 implementation, never a cryptographic mock. An existing checkout
# can be supplied to avoid downloading it during local iteration.
set(TRAIL_MATE_CRYPTO_SOURCE_DIR "" CACHE PATH "Directory containing Crypto SHA256.cpp")
if(NOT TRAIL_MATE_CRYPTO_SOURCE_DIR)
    include(FetchContent)
    if(POLICY CMP0135)
        cmake_policy(SET CMP0135 NEW)
    endif()
    FetchContent_Declare(reticulum_discovery_crypto_source
        URL "https://codeload.github.com/rweather/arduinolibs/tar.gz/37a76b8f7516568e1c575b6dc9268da1ccaac6b6")
    FetchContent_MakeAvailable(reticulum_discovery_crypto_source)
    set(TRAIL_MATE_CRYPTO_SOURCE_DIR "${reticulum_discovery_crypto_source_SOURCE_DIR}/libraries/Crypto")
endif()
add_library(reticulum_discovery_crypto STATIC
    "${TRAIL_MATE_CRYPTO_SOURCE_DIR}/SHA256.cpp"
    "${TRAIL_MATE_CRYPTO_SOURCE_DIR}/Hash.cpp"
    "${TRAIL_MATE_CRYPTO_SOURCE_DIR}/Crypto.cpp")
target_include_directories(reticulum_discovery_crypto PUBLIC "${TRAIL_MATE_CRYPTO_SOURCE_DIR}")
add_executable(reticulum_discovery_stamp test_discovery_stamp.cpp
    "${REPO}/platform/esp/arduino_common/src/chat/infra/reticulum/discovery_stamp_verifier.cpp")
target_compile_features(reticulum_discovery_stamp PRIVATE cxx_std_17)
target_compile_definitions(reticulum_discovery_stamp PRIVATE ESP_PLATFORM)
target_include_directories(reticulum_discovery_stamp PRIVATE
    "${CMAKE_CURRENT_LIST_DIR}/fakes" "${REPO}/platform/esp/arduino_common/include")
target_link_libraries(reticulum_discovery_stamp PRIVATE reticulum_discovery_crypto)
