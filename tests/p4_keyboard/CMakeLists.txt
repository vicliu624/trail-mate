cmake_minimum_required(VERSION 3.20)
project(p4_keyboard_contract LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(REPO "${CMAKE_CURRENT_LIST_DIR}/../..")
set(LVGL_DIR "" CACHE PATH "LVGL source directory")
set(LVGL_LIBRARY "" CACHE FILEPATH "Compatible prebuilt LVGL library")
if(LVGL_LIBRARY)
    add_library(lvgl STATIC IMPORTED)
    set_target_properties(lvgl PROPERTIES IMPORTED_LOCATION "${LVGL_LIBRARY}")
    target_include_directories(lvgl INTERFACE "${LVGL_DIR}" "${REPO}/tests/contacts_card_view")
    target_compile_definitions(lvgl INTERFACE LV_CONF_INCLUDE_SIMPLE)
else()
    set(LV_BUILD_CONF_PATH "${REPO}/tests/contacts_card_view/lv_conf.h" CACHE PATH "" FORCE)
    set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
    set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    add_subdirectory("${LVGL_DIR}" lvgl)
endif()
enable_testing()
foreach(tick_rate IN ITEMS 100 1000)
    set(target "p4_keyboard_contract_${tick_rate}hz")
    add_executable(${target} test_keyboard.cpp)
    target_include_directories(${target} PRIVATE fakes
        "${REPO}/platform/esp/idf_components/t_display_p4/include")
    target_link_libraries(${target} PRIVATE lvgl)
    target_compile_definitions(${target} PRIVATE configTICK_RATE_HZ=${tick_rate})
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${target} COMMAND ${target})
endforeach()
