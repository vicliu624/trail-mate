# Compile production lifecycle callbacks with a minimal Settings/IME fixture.
# LVGL groups, objects, deletion and input dispatch are not mocked.
set(COMPONENTS "${REPO}/modules/ui_shared/src/ui/screens/settings/settings_page_components.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${COMPONENTS}")
file(READ "${COMPONENTS}" SOURCE)
function(extract_between START END OUTPUT)
    string(FIND "${SOURCE}" "${START}" FIRST)
    if(FIRST LESS 0)
        message(FATAL_ERROR "Settings lifecycle extraction markers changed")
    endif()
    string(SUBSTRING "${SOURCE}" ${FIRST} -1 TAIL)
    string(FIND "${TAIL}" "${END}" LENGTH)
    if(LENGTH LESS 0)
        message(FATAL_ERROR "Settings lifecycle end marker changed")
    endif()
    string(SUBSTRING "${TAIL}" 0 ${LENGTH} BODY)
    set(${OUTPUT} "${BODY}" PARENT_SCOPE)
endfunction()
extract_between("static void modal_prepare_group(" "static void modal_add_focus_obj(" MODAL)
extract_between("static void settings_back_cb(" "static void bind_dynamic_option_storage_to_items(" BACK)
extract_between("static lv_obj_t* create_modal_root(" "static void on_text_save_clicked(" LAYOUT)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/settings_lifecycle_actual.inc" "${MODAL}\n${BACK}\n${LAYOUT}")
add_executable(settings_navigation test_navigation.cpp)

else()

endif()
