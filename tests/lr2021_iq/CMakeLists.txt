cmake_minimum_required(VERSION 3.20)
project(lr2021_iq_rx_contract LANGUAGES C CXX)
enable_testing()
include(../../cmake/TrailMateLr2021Iq.cmake)
trailmate_add_lr2021_iq(trailmate_iq_rx)
get_target_property(IQ_SOURCE trailmate_iq_rx TRAIL_MATE_SOURCE_DIR)
add_executable(p4_iq_wait_test test_p4_iq_wait.cpp)
target_include_directories(p4_iq_wait_test PRIVATE wait_fakes ../../platform/esp/idf_common/include)
target_compile_features(p4_iq_wait_test PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(p4_iq_wait_test PRIVATE /UNDEBUG)
else()
    target_compile_options(p4_iq_wait_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME p4_iq_wait COMMAND p4_iq_wait_test)
set_tests_properties(p4_iq_wait PROPERTIES TIMEOUT 5)
add_executable(iq_stream_cancel_test test_stream_cancel.c)
target_link_libraries(iq_stream_cancel_test PRIVATE trailmate_iq_rx)
target_compile_definitions(iq_stream_cancel_test PRIVATE
    TRAIL_MATE_UPSTREAM_RADIO_TEST="${IQ_SOURCE}/tests/test_radio.c")
add_test(NAME iq_stream_cancel COMMAND iq_stream_cancel_test)
set_tests_properties(iq_stream_cancel PROPERTIES TIMEOUT 5)
# Compile the exact generated production C source through its ESP path on the
# host. This is a stack/ownership contract test, not an ESP firmware build.
add_library(trailmate_iq_rx_esp_contract STATIC
    "${CMAKE_CURRENT_BINARY_DIR}/trailmate_iq_rx-rx-overlay/lr2021_iq.c")
get_target_property(IQ_INCLUDE_DIRS trailmate_iq_rx INTERFACE_INCLUDE_DIRECTORIES)
target_include_directories(trailmate_iq_rx_esp_contract PUBLIC ${IQ_INCLUDE_DIRS})
target_compile_definitions(trailmate_iq_rx_esp_contract PRIVATE ESP_PLATFORM=1)
target_compile_features(trailmate_iq_rx_esp_contract PRIVATE c_std_11)
if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
    target_compile_options(trailmate_iq_rx_esp_contract PRIVATE -Wframe-larger-than=512 -Werror=frame-larger-than=512)
endif()
add_executable(iq_stream_esp_storage_test test_stream_cancel.c)
target_compile_definitions(iq_stream_esp_storage_test PRIVATE TRAIL_MATE_TEST_ESP_STREAM=1
    TRAIL_MATE_UPSTREAM_RADIO_TEST="${IQ_SOURCE}/tests/test_radio.c")
target_link_libraries(iq_stream_esp_storage_test PRIVATE trailmate_iq_rx_esp_contract)
add_test(NAME iq_stream_esp_storage COMMAND iq_stream_esp_storage_test)
set_tests_properties(iq_stream_esp_storage PROPERTIES TIMEOUT 5)
foreach(kind radio queue)
    add_executable(iq_${kind}_test "${IQ_SOURCE}/tests/test_${kind}.c")
    target_link_libraries(iq_${kind}_test PRIVATE trailmate_iq_rx)
    target_compile_features(iq_${kind}_test PRIVATE c_std_11)
    if(NOT WIN32 AND kind STREQUAL "queue")
        find_package(Threads REQUIRED)
        target_link_libraries(iq_${kind}_test PRIVATE Threads::Threads)
    endif()
    add_test(NAME iq_${kind} COMMAND iq_${kind}_test)
endforeach()

# Exercise the actual P4 adapter methods, extracted without rewriting them.
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(P4_DRIVER "${CMAKE_CURRENT_LIST_DIR}/../../platform/esp/idf_common/src/lr2021_radio.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${P4_DRIVER}" extract_transport.py)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract_transport.py"
    "${P4_DRIVER}" "${CMAKE_CURRENT_BINARY_DIR}/p4_iq_transport.inc" iqFifoRead
    COMMAND_ERROR_IS_FATAL ANY)
add_executable(p4_iq_transport_test test_transport.cpp)
target_include_directories(p4_iq_transport_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(p4_iq_transport_test PRIVATE trailmate_iq_rx)
target_compile_features(p4_iq_transport_test PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(p4_iq_transport_test PRIVATE /UNDEBUG)
else()
    target_compile_options(p4_iq_transport_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME p4_iq_transport COMMAND p4_iq_transport_test)
add_executable(p4_iq_runtime_test test_runtime.cpp
    ../../platform/esp/idf_common/src/platform_ui_radio_iq_runtime.cpp)
target_compile_features(p4_iq_runtime_test PRIVATE cxx_std_17)
target_compile_definitions(p4_iq_runtime_test PRIVATE ESP_PLATFORM=1 TRAIL_MATE_IDF_RADIO_LR2021=1)
target_include_directories(p4_iq_runtime_test PRIVATE runtime_fakes
    ../../modules/core_sys/include ../../platform/esp/idf_common/include)
if(WIN32 AND MINGW)
    # This legacy MinGW lacks gthreads. The task fixture is sequential;
    # Linux CI and the ESP firmware use their actual standard-library mutex.
    target_include_directories(p4_iq_runtime_test PRIVATE ../p4_hardware/fakes)
endif()
if(MSVC)
    target_compile_options(p4_iq_runtime_test PRIVATE /UNDEBUG)
else()
    target_compile_options(p4_iq_runtime_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME p4_iq_runtime COMMAND p4_iq_runtime_test)
