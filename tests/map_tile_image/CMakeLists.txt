cmake_minimum_required(VERSION 3.20)
project(map_tile_image LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(REPO "${CMAKE_CURRENT_LIST_DIR}/../..")
if(NOT TARGET lvgl)
    set(LVGL_DIR "" CACHE PATH "Pinned LVGL 9.4.0 source directory")
    set(LV_BUILD_CONF_PATH "${REPO}/tests/map_poi_view/lv_conf.h" CACHE PATH "" FORCE)
    set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
    set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
    add_subdirectory(${LVGL_DIR} lvgl)
endif()
target_compile_definitions(lvgl PUBLIC LV_USE_LODEPNG=1)
add_executable(map_tile_image test_image.cpp)
target_include_directories(map_tile_image PRIVATE "${REPO}/platform/esp/arduino_common/include")
target_link_libraries(map_tile_image PRIVATE lvgl)
if(MSVC)
    target_compile_options(map_tile_image PRIVATE /UNDEBUG)
else()
    target_compile_options(map_tile_image PRIVATE -UNDEBUG -Wall -Wextra)
endif()
enable_testing()
add_test(NAME map_tile_image COMMAND map_tile_image)
set_tests_properties(map_tile_image PROPERTIES TIMEOUT 60)
